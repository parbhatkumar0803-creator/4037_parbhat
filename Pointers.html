Question 2: Write a program to swap two numbers using pointers.

Code


#include <stdio.h>
#include <conio.h>      
void swap(int *p1, int *p2) {
    int res = *p1;
    *p1 = *p2;
    *p2 = res;
}

void main() {
    int x = 20, y = 10;

    clrscr();
    printf("Before swapping: x = %d, y = %d\n", x, y);
    swap(&x, &y);
    printf("After swapping: x = %d, y = %d\n", x, y);

}

Output


Before swapping: x = 20, y = 10
After swapping: x = 10, y = 20

Question 4: Write a program in which we are having 2d array of 2 by 3. Insert the value through pointer and print the values.

Code


#include <stdio.h>
#include 
#include     
void main()
{
    int arr[2][3], *ptr, i, j;

    ptr = arr[0];

    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j <= 2; j++)
        {
            printf("Enter the value: ");
            scanf("%d", &arr[i][j]);
        }
    }

    for(i = 0; i <= 1; i++)
    {
        for(j = 0; j <= 2; j++)
        {
            printf("%d\t", *ptr);
            ptr++;
        }

    }
delay(2000);

}
